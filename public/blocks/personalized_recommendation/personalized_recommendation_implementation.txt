// This file is part of Moodle - http://moodle.org/

/**
 * Fetches and renders the Next Best Activity card. Uses DOM APIs only (textContent),
 * so reasons or names can never inject HTML.
 *
 * @module     block_personalized_recommendation/sidebar
 * @copyright  2026
 * @license    http://www.gnu.org/copyleft/gpl.html GNU GPL v3 or later
 */
define(['core/ajax', 'core/str'], function(Ajax, Str) {
    var COMPONENT = 'block_personalized_recommendation';
    var KEYS = ['start', 'review', 'alsoconsider', 'minutes', 'unavailable'];
    var labels = {};

    var el = function(tag, className, text) {
        var node = document.createElement(tag);
        if (className) {
            node.className = className;
        }
        if (text !== undefined && text !== null) {
            node.textContent = text;
        }
        return node;
    };

    var meta = function(item) {
        var parts = [item.typelabel];
        if (item.topic) {
            parts.push(item.topic);
        }
        if (item.minutes !== null && item.minutes !== undefined) {
            parts.push(labels.minutes.replace('{$a}', item.minutes));
        }
        return parts.join(' \u00B7 ');
    };

    var renderHero = function(item) {
        var hero = el('div', 'block-pr-hero');
        hero.appendChild(el('div', 'block-pr-name', item.name));
        hero.appendChild(el('div', 'block-pr-meta small', meta(item)));
        if (item.isreview) {
            hero.appendChild(el('span', 'badge bg-secondary block-pr-badge', labels.review));
        }
        var start = el('a', 'btn btn-primary w-100 block-pr-start', labels.start);
        start.href = item.url;
        var wrap = el('div');
        wrap.appendChild(hero);
        wrap.appendChild(start);
        if (item.reason) {
            wrap.appendChild(el('p', 'block-pr-reason small', item.reason));
        }
        return wrap;
    };

    var renderOthers = function(items) {
        var section = el('div', 'block-pr-others');
        section.appendChild(el('div', 'block-pr-subheading small fw-bold', labels.alsoconsider));
        var list = el('ul', 'list-unstyled mb-0');
        items.forEach(function(item) {
            var li = el('li', 'block-pr-other');
            var a = el('a', null, item.name);
            a.href = item.url;
            li.appendChild(a);
            li.appendChild(el('div', 'small text-muted', meta(item)));
            list.appendChild(li);
        });
        section.appendChild(list);
        return section;
    };

    var render = function(box, data) {
        box.textContent = '';
        box.removeAttribute('aria-busy');
        if (data.status !== 'ok' || !data.items.length) {
            box.appendChild(el('p', 'text-muted small mb-0', data.message || labels.unavailable));
            return;
        }
        box.appendChild(renderHero(data.items[0]));
        if (data.items.length > 1) {
            box.appendChild(renderOthers(data.items.slice(1)));
        }
    };

    return {
        /**
         * @param {Object} config {rootid, courseid}
         */
        init: function(config) {
            var root = document.getElementById(config.rootid);
            if (!root) {
                return;
            }
            var box = root.querySelector('[data-region="content"]');
            Str.get_strings(KEYS.map(function(key) {
                return {key: key, component: COMPONENT};
            })).then(function(strings) {
                KEYS.forEach(function(key, i) {
                    labels[key] = strings[i];
                });
                return Ajax.call([{
                    methodname: 'block_personalized_recommendation_get_next_activity',
                    args: {courseid: config.courseid}
                }])[0];
            }).then(function(data) {
                render(box, data);
                return data;
            }).catch(function() {
                render(box, {status: 'unavailable', items: [], message: labels.unavailable || ''});
            });
        }
    };
});

// This file is part of Moodle - http://moodle.org/

/**
 * Student-facing "Next Best Activity" block.
 *
 * The block only renders a shell. The recommendation is fetched asynchronously
 * (see amd/src/sidebar.js) so a slow or offline analytics API can never delay
 * or break the course page.
 *
 * @package    block_personalized_recommendation
 * @license    http://www.gnu.org/copyleft/gpl.html GNU GPL v3 or later
 */

defined('MOODLE_INTERNAL') || die();

/**
 * Block class.
 */
class block_personalized_recommendation extends block_base {
    /**
     * Set the block title.
     */
    public function init(): void {
        $this->title = get_string('pluginname', 'block_personalized_recommendation');
    }

    /**
     * Only inside courses.
     *
     * @return array
     */
    public function applicable_formats(): array {
        return ['course-view' => true, 'mod' => true, 'site' => false, 'my' => false];
    }

    /**
     * One card per course page is enough.
     *
     * @return bool
     */
    public function instance_allow_multiple(): bool {
        return false;
    }

    /**
     * The block has site-level settings (settings.php).
     *
     * @return bool
     */
    public function has_config(): bool {
        return true;
    }

    /**
     * Build the block content (shell only; data arrives via AJAX).
     *
     * @return stdClass
     */
    public function get_content(): stdClass {
        global $OUTPUT;

        if ($this->content !== null) {
            return $this->content;
        }
        $this->content = new stdClass();
        $this->content->text = '';
        $this->content->footer = '';

        if (!isloggedin() || isguestuser()) {
            return $this->content;
        }
        $courseid = (int) $this->page->course->id;
        $context = context_course::instance($courseid);
        if (!has_capability('block/personalized_recommendation:view', $context)) {
            return $this->content;
        }

        $rootid = 'block-pr-' . $this->instance->id;
        $this->page->requires->js_call_amd('block_personalized_recommendation/sidebar', 'init', [[
            'rootid' => $rootid,
            'courseid' => $courseid,
        ]]);
        $this->content->text = $OUTPUT->render_from_template('block_personalized_recommendation/sidebar', [
            'rootid' => $rootid,
        ]);
        return $this->content;
    }
}

<?php
// This file is part of Moodle - http://moodle.org/

namespace block_personalized_recommendation;

use core\http_client;
use GuzzleHttp\ClientInterface;
use GuzzleHttp\Exception\GuzzleException;

/**
 * Server-side client for the Python recommendation API.
 *
 * Uses core\http_client, so Moodle's proxy and "HTTP security" settings
 * (blocked hosts / allowed ports) apply. The token never reaches the browser
 * and is never included in exception messages.
 *
 * @package    block_personalized_recommendation
 * @license    http://www.gnu.org/copyleft/gpl.html GNU GPL v3 or later
 */
class api_client {
    /** @var ClientInterface */
    private $http;

    /**
     * @param ClientInterface|null $http Injectable for tests; defaults to a client built from plugin settings.
     */
    public function __construct(?ClientInterface $http = null) {
        $this->http = $http ?? self::default_http();
    }

    /**
     * Build the HTTP client from admin settings.
     *
     * @return ClientInterface
     */
    private static function default_http(): ClientInterface {
        $url = trim((string) get_config('block_personalized_recommendation', 'api_url'));
        $token = (string) get_config('block_personalized_recommendation', 'api_token');
        if ($url === '' || $token === '') {
            throw new \moodle_exception('apinotconfigured', 'block_personalized_recommendation');
        }
        $timeout = (int) get_config('block_personalized_recommendation', 'timeout');
        return new http_client([
            'base_uri' => rtrim($url, '/') . '/',
            'timeout' => $timeout > 0 ? $timeout : 4,
            'connect_timeout' => 2,
            'http_errors' => false,
            'headers' => ['X-API-Token' => $token, 'Accept' => 'application/json'],
        ]);
    }

    /**
     * GET a JSON resource.
     *
     * @param string $path e.g. "recommend/3"
     * @param array $query query-string parameters
     * @return array decoded JSON object
     */
    public function get(string $path, array $query = []): array {
        return $this->request('GET', $path, $query ? ['query' => $query] : []);
    }

    /**
     * Perform the request and normalise every failure into a moodle_exception.
     *
     * @param string $method
     * @param string $path
     * @param array $options Guzzle request options
     * @return array
     */
    private function request(string $method, string $path, array $options = []): array {
        try {
            $response = $this->http->request($method, ltrim($path, '/'), $options);
        } catch (GuzzleException $e) {
            debugging('Analytics API request failed: ' . $e->getMessage(), DEBUG_DEVELOPER);
            throw new \moodle_exception('apiunavailable', 'block_personalized_recommendation');
        }

        $status = $response->getStatusCode();
        if ($status === 401) {
            throw new \moodle_exception('apiauth', 'block_personalized_recommendation');
        }
        if ($status === 409) {
            throw new \moodle_exception('apibusy', 'block_personalized_recommendation');
        }
        if ($status < 200 || $status >= 300) {
            throw new \moodle_exception('apierror', 'block_personalized_recommendation', '', $status);
        }

        try {
            $data = json_decode((string) $response->getBody(), true, 512, JSON_THROW_ON_ERROR);
        } catch (\JsonException $e) {
            throw new \moodle_exception('apibadresponse', 'block_personalized_recommendation');
        }
        if (!is_array($data)) {
            throw new \moodle_exception('apibadresponse', 'block_personalized_recommendation');
        }
        return $data;
    }
}

namespace block_personalized_recommendation;

defined('MOODLE_INTERNAL') || die();

/**
 * Ensures every course has the student recommendation block in its side drawer.
 *
 * @package    block_personalized_recommendation
 * @license    http://www.gnu.org/copyleft/gpl.html GNU GPL v3 or later
 */
class block_provisioner {
    /** Provision this block on all existing non-site courses. */
    public static function add_to_existing_courses(): void {
        global $DB;

        upgrade_set_timeout(3600);
        $courses = $DB->get_recordset_select('course', 'id <> :siteid', ['siteid' => SITEID], '', 'id');
        foreach ($courses as $course) {
            self::ensure_for_course((int) $course->id);
        }
        $courses->close();
    }

    /**
     * Add one inherited side-drawer block to a course when none exists.
     *
     * @param int $courseid Course id.
     */
    public static function ensure_for_course(int $courseid): void {
        global $DB;

        if ($courseid === SITEID) {
            return;
        }

        $context = \context_course::instance($courseid);
        $exists = $DB->record_exists('block_instances', [
            'blockname' => 'personalized_recommendation',
            'parentcontextid' => $context->id,
            'showinsubcontexts' => 1,
            'pagetypepattern' => '*',
        ]);
        if ($exists) {
            return;
        }

        $page = new \moodle_page();
        $page->set_context($context);
        $page->set_pagelayout('incourse');
        $page->blocks->add_region('side-pre', false);
        $page->blocks->add_block(
            'personalized_recommendation',
            'side-pre',
            0,
            true,
            '*'
        );
    }
}

<?php
// This file is part of Moodle - http://moodle.org/

namespace block_personalized_recommendation\external;

use block_personalized_recommendation\api_client;
use core_external\external_api;
use core_external\external_function_parameters;
use core_external\external_multiple_structure;
use core_external\external_single_structure;
use core_external\external_value;

defined('MOODLE_INTERNAL') || die();

/**
 * Returns the Next Best Activity list for the CURRENT user.
 *
 * The learner id is never a parameter: it is always $USER->id, so a student
 * cannot ask for someone else's recommendations. Only activities the user can
 * actually open (visible, available, in this course) are returned, each with its
 * real Moodle URL.
 *
 * @package    block_personalized_recommendation
 * @license    http://www.gnu.org/copyleft/gpl.html GNU GPL v3 or later
 */
class get_next_activity extends external_api {
    /** Fetch enough ranked candidates to replace activities already completed in Moodle. */
    private const CANDIDATE_LIMIT = 20;

    /**
     * @return external_function_parameters
     */
    public static function execute_parameters(): external_function_parameters {
        return new external_function_parameters([
            'courseid' => new external_value(PARAM_INT, 'Course id'),
        ]);
    }

    /**
     * @param int $courseid
     * @return array
     */
    public static function execute(int $courseid): array {
        global $USER;
        ['courseid' => $courseid] = self::validate_parameters(self::execute_parameters(), ['courseid' => $courseid]);

        $context = \context_course::instance($courseid);
        self::validate_context($context);
        require_capability('block/personalized_recommendation:view', $context);

        $result = ['status' => 'ok', 'message' => '', 'topic' => '', 'items' => []];

        // Staff previewing the course are not learners: do not invent a recommendation for them.
        if (!has_capability('moodle/course:isincompletionreports', $context)) {
            return ['status' => 'notlearner', 'message' => get_string('notlearner', 'block_personalized_recommendation')] + $result;
        }

        $max = max(1, min(5, (int) get_config('block_personalized_recommendation', 'maxitems') ?: 3));
        $method = (string) (get_config('block_personalized_recommendation', 'method') ?: 'rule_based');

        $completed = self::get_completed_course_module_ids($courseid, (int) $USER->id);
        try {
            $payload = (new api_client())->get('recommend/' . (int) $USER->id, [
                'k' => self::CANDIDATE_LIMIT,
                'method' => $method,
            ]);
        } catch (\moodle_exception $e) {
            return ['status' => 'unavailable', 'message' => get_string('unavailable', 'block_personalized_recommendation')] + $result;
        }

        $modinfo = get_fast_modinfo($courseid);
        $items = [];
        foreach (($payload['recommendations'] ?? []) as $rec) {
            $cmid = (int) ($rec['course_module_id'] ?? 0);
            if (isset($completed[$cmid])) {
                continue;
            }
            $cm = $modinfo->cms[$cmid] ?? null;
            if (!$cm || !$cm->uservisible || !$cm->url) {
                continue;
            }
            $items[] = [
                'name' => $cm->get_formatted_name(),
                'url' => $cm->url->out(false),
                'typelabel' => get_string('modulename', 'mod_' . $cm->modname),
                'topic' => (string) ($rec['topic'] ?? ''),
                // The API prefixes "[method had no pick; X fallback]" for evaluation; students need only the explanation.
                'reason' => trim((string) preg_replace('/^\[[^\]]*\]\s*/', '', (string) ($rec['reason'] ?? ''))),
                'isreview' => !empty($rec['is_review']),
                'minutes' => isset($rec['est_minutes']) ? (int) $rec['est_minutes'] : null,
            ];
            if (count($items) >= $max) {
                break;
            }
        }

        if (!$items) {
            return ['status' => 'empty', 'message' => get_string('empty', 'block_personalized_recommendation')] + $result;
        }
        $result['items'] = $items;
        $result['topic'] = (string) ($payload['focus']['topic'] ?? '');
        return $result;
    }

    /**
     * Find course activities the learner has completed.
     *
     * Completion records are authoritative for suppressing recommendations.
     * Merely opening an activity does not make it ineligible.
     *
     * @param int $courseid Course id.
     * @param int $userid Learner id.
     * @return array<int, bool> Course-module ids keyed for constant-time lookup.
     */
    private static function get_completed_course_module_ids(int $courseid, int $userid): array {
        global $DB;

        $completed = $DB->get_fieldset_sql(
            "SELECT DISTINCT cmc.coursemoduleid
               FROM {course_modules_completion} cmc
               JOIN {course_modules} cm ON cm.id = cmc.coursemoduleid
              WHERE cm.course = :courseid
                AND cmc.userid = :userid
                AND cmc.completionstate IN (:complete, :passed, :failed)",
            [
                'courseid' => $courseid,
                'userid' => $userid,
                'complete' => COMPLETION_COMPLETE,
                'passed' => COMPLETION_COMPLETE_PASS,
                'failed' => COMPLETION_COMPLETE_FAIL,
            ]
        );

        $completedids = [];
        foreach ($completed as $cmid) {
            $completedids[(int) $cmid] = true;
        }
        return $completedids;
    }

    /**
     * @return external_single_structure
     */
    public static function execute_returns(): external_single_structure {
        return new external_single_structure([
            'status' => new external_value(PARAM_ALPHA, 'ok, empty, unavailable or notlearner'),
            'message' => new external_value(PARAM_TEXT, 'Message to show instead of items', VALUE_DEFAULT, ''),
            'topic' => new external_value(PARAM_TEXT, 'Topic the learner should work on', VALUE_DEFAULT, ''),
            'items' => new external_multiple_structure(new external_single_structure([
                'name' => new external_value(PARAM_TEXT, 'Activity name'),
                'url' => new external_value(PARAM_URL, 'Activity URL'),
                'typelabel' => new external_value(PARAM_TEXT, 'Localised activity type'),
                'topic' => new external_value(PARAM_TEXT, 'Topic'),
                'reason' => new external_value(PARAM_TEXT, 'Why this activity'),
                'isreview' => new external_value(PARAM_BOOL, 'Learner has already used this activity'),
                'minutes' => new external_value(PARAM_INT, 'Estimated minutes', VALUE_DEFAULT, null, NULL_ALLOWED),
            ])),
        ]);
    }
}

<?php

defined('MOODLE_INTERNAL') || die();

/**
 * Upgrade the recommendation block plugin.
 *
 * @param int $oldversion Previous plugin version.
 * @param stdClass $block Plugin record.
 * @return bool
 */
function xmldb_block_personalized_recommendation_upgrade($oldversion, $block): bool {
    if ($oldversion < 2026101800) {
        \block_personalized_recommendation\block_provisioner::add_to_existing_courses();
        upgrade_plugin_savepoint(true, 2026101800, 'block', 'personalized_recommendation');
    }
    if ($oldversion < 2026101802) {
        upgrade_plugin_savepoint(true, 2026101802, 'block', 'personalized_recommendation');
    }
    return true;
}
